cmake_minimum_required(VERSION 3.22.1)
project(airplay_native C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT UXPLAY_SOURCE)
    message(FATAL_ERROR "Receive-core sources are required")
endif()
include(cmake/ReceiverCore.cmake)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJCXX)
    add_library(airplay_player STATIC player/player.cpp)
else()
    add_library(airplay_player SHARED player/player.cpp)
endif()
set_target_properties(airplay_player PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
target_include_directories(airplay_player PUBLIC player)
target_link_libraries(airplay_player PRIVATE receiver_core)
target_compile_definitions(receiver_core PRIVATE TARGET_RT_LITTLE_ENDIAN=1 RAOP_CN="1,2,3")
if(ANDROID)
    if(NOT DEPS_SOURCE)
        message(FATAL_ERROR "Pinned Android media sources are required")
    endif()
    set(OBOE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(OBOE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(${DEPS_SOURCE}/oboe ${CMAKE_BINARY_DIR}/oboe)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../vendor/alac ${CMAKE_BINARY_DIR}/alac)
    target_sources(airplay_player PRIVATE player/android_audio.cpp player/android_video.cpp player/android_audio_decoder.cpp)
    target_link_libraries(airplay_player PRIVATE alac_decoder oboe mediandk android log m dl)
    target_link_options(airplay_player PRIVATE "-Wl,-z,max-page-size=16384")
elseif(WIN32)
    include(${CMAKE_CURRENT_SOURCE_DIR}/../windows/cmake/WindowsNative.cmake)
elseif(APPLE)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        target_sources(airplay_player PRIVATE player/ios_audio.mm)
        target_link_libraries(airplay_player PRIVATE "-framework AVFAudio")
    else()
        target_sources(airplay_player PRIVATE player/macos_audio.cpp)
    endif()
    target_sources(airplay_player PRIVATE player/macos_video.cpp player/macos_audio_decoder.cpp)
    target_link_libraries(airplay_player PRIVATE "-framework AudioToolbox" "-framework CoreAudio"
        "-framework CoreMedia" "-framework CoreVideo" "-framework VideoToolbox" "-framework CoreFoundation")
    set_target_properties(airplay_player PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(${CMAKE_CURRENT_SOURCE_DIR}/../linux/cmake/LinuxPlayer.cmake)
    airplay_configure_linux_player(airplay_player)
endif()
option(AIRPLAY_BUILD_TESTS "Build shared playback regressions" OFF)
if(AIRPLAY_BUILD_TESTS AND APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_testing()
    add_executable(player_tests player-tests/main.cpp)
    target_include_directories(player_tests PRIVATE player)
    target_link_libraries(player_tests PRIVATE airplay_player "-framework CoreVideo")
    add_test(NAME playback COMMAND player_tests)
    add_executable(session_tests player-tests/session_test.cpp)
    target_include_directories(session_tests PRIVATE player)
    target_link_libraries(session_tests PRIVATE airplay_player receiver_core "-framework CoreVideo")
    add_test(NAME sender_resume COMMAND session_tests)
endif()
