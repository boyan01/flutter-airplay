cmake_minimum_required(VERSION 3.22.1)
project(airplay_native C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT UXPLAY_SOURCE OR NOT PLIST_SOURCE OR NOT CRYPTO_PREFIX OR NOT DEPS_SOURCE)
    message(FATAL_ERROR "Pinned receive-core, crypto and media sources are required")
endif()
include(cmake/ReceiverCore.cmake)
include(cmake/BuildFFmpeg.cmake)
add_library(airplay_player SHARED player/player.cpp)
set_target_properties(airplay_player PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
target_include_directories(airplay_player PUBLIC player)
target_link_libraries(airplay_player PRIVATE receiver_core ffmpeg::avcodec)
add_dependencies(airplay_player ffmpeg_ep)
target_compile_definitions(receiver_core PRIVATE TARGET_RT_LITTLE_ENDIAN=1 RAOP_CN="1,2,3")
if(ANDROID)
    set(OBOE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(OBOE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(${DEPS_SOURCE}/oboe ${CMAKE_BINARY_DIR}/oboe)
    target_sources(airplay_player PRIVATE player/android_audio.cpp player/android_video.cpp)
    target_link_libraries(airplay_player PRIVATE oboe mediandk android log m dl)
    target_link_options(airplay_player PRIVATE "-Wl,-z,max-page-size=16384")
elseif(APPLE)
    target_sources(airplay_player PRIVATE player/macos_audio.cpp player/macos_video.cpp)
    target_link_libraries(airplay_player PRIVATE "-framework AudioToolbox" "-framework CoreAudio"
        "-framework CoreMedia" "-framework CoreVideo" "-framework VideoToolbox" "-framework CoreFoundation")
    set_target_properties(airplay_player PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
endif()
option(AIRPLAY_BUILD_TESTS "Build shared playback regressions" OFF)
if(AIRPLAY_BUILD_TESTS)
    enable_testing()
    add_executable(player_tests player-tests/main.cpp)
    target_include_directories(player_tests PRIVATE player)
    target_link_libraries(player_tests PRIVATE airplay_player ffmpeg::avcodec "-framework CoreVideo")
    add_test(NAME playback COMMAND player_tests)
    add_executable(session_tests player-tests/session_test.cpp)
    target_include_directories(session_tests PRIVATE player)
    target_link_libraries(session_tests PRIVATE airplay_player receiver_core ffmpeg::avcodec "-framework CoreVideo")
    add_test(NAME sender_resume COMMAND session_tests)
endif()
