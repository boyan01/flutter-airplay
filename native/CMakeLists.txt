cmake_minimum_required(VERSION 3.22.1)
project(airplay_native C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(AIRPLAY_CONTROL_ONLY "Build receiver control tests without Dart or media backends" OFF)
if(NOT UXPLAY_SOURCE AND NOT AIRPLAY_CONTROL_ONLY)
    message(FATAL_ERROR "Receive-core sources are required")
endif()
include(cmake/Plist.cmake)
add_library(airplay_receiver_control OBJECT receiver/receiver.cpp)
target_link_libraries(airplay_receiver_control PRIVATE plist)
if(AIRPLAY_CONTROL_ONLY)
    if(WIN32)
        include(${CMAKE_CURRENT_SOURCE_DIR}/../windows/cmake/WindowsNative.cmake)
    endif()
    enable_testing()
    add_subdirectory(tests/receiver)
    return()
endif()
include(cmake/ReceiverCore.cmake)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_library(airplay_player STATIC playback/player.cpp)
else()
    add_library(airplay_player SHARED playback/player.cpp)
endif()
add_library(airplay_protocol OBJECT protocol/uxplay_adapter.cpp)
target_link_libraries(airplay_protocol PRIVATE receiver_core)
target_sources(airplay_player PRIVATE $<TARGET_OBJECTS:airplay_receiver_control> $<TARGET_OBJECTS:airplay_protocol>)
option(AIRPLAY_BUILD_FFI "Build the Dart transport adapter" ON)
if(AIRPLAY_BUILD_FFI)
# NativeApi.postCObject supplies the VM function pointer at runtime. Only the
# SDK's public CObject layout is needed at build time; no VM library is linked.
find_program(AIRPLAY_DART_EXECUTABLE NAMES dart dart.bat
    HINTS "${FLUTTER_ROOT}/bin" "$ENV{FLUTTER_ROOT}/bin"
    NO_CMAKE_FIND_ROOT_PATH)
if(AIRPLAY_DART_EXECUTABLE)
    get_filename_component(airplay_dart_real "${AIRPLAY_DART_EXECUTABLE}" REALPATH)
    get_filename_component(airplay_dart_bin "${airplay_dart_real}" DIRECTORY)
endif()
find_path(AIRPLAY_DART_INCLUDE dart_native_api.h
    HINTS "${DART_SDK_ROOT}/include" "$ENV{DART_SDK_ROOT}/include"
        "${FLUTTER_ROOT}/bin/cache/dart-sdk/include" "$ENV{FLUTTER_ROOT}/bin/cache/dart-sdk/include"
        "${airplay_dart_bin}/cache/dart-sdk/include" "${airplay_dart_bin}/../include"
    NO_CMAKE_FIND_ROOT_PATH REQUIRED)
add_library(airplay_ffi OBJECT ffi/receiver_ffi.cpp)
target_include_directories(airplay_ffi PRIVATE "${AIRPLAY_DART_INCLUDE}")
target_link_libraries(airplay_ffi PRIVATE plist)
target_sources(airplay_player PRIVATE $<TARGET_OBJECTS:airplay_ffi>)
endif()
set_target_properties(airplay_player PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
target_include_directories(airplay_player PUBLIC include/airplay PRIVATE playback)
target_link_libraries(airplay_player PRIVATE receiver_core)
target_compile_definitions(receiver_core PRIVATE TARGET_RT_LITTLE_ENDIAN=1 RAOP_CN="1,2,3")
if(ANDROID)
    if(NOT DEPS_SOURCE)
        message(FATAL_ERROR "Pinned Android media sources are required")
    endif()
    set(OBOE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(OBOE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(${DEPS_SOURCE}/oboe ${CMAKE_BINARY_DIR}/oboe)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../vendor/alac ${CMAKE_BINARY_DIR}/alac)
    target_sources(airplay_player PRIVATE backends/android/android_audio.cpp backends/android/android_video.cpp backends/android/android_audio_decoder.cpp)
    target_link_libraries(airplay_player PRIVATE alac_decoder oboe mediandk android log m dl)
    target_link_options(airplay_player PRIVATE "-Wl,-z,max-page-size=16384")
    option(AIRPLAY_ANDROID_BUILD_TESTS "Build Android device playback fixtures" OFF)
    if(AIRPLAY_ANDROID_BUILD_TESTS)
        add_library(player_regression SHARED
            ../android/tests/player/player_test.cpp tests/playback/session_test.cpp)
        target_link_libraries(player_regression PRIVATE airplay_player receiver_core oboe android mediandk log)
        target_link_options(player_regression PRIVATE "-Wl,-z,max-page-size=16384" -static-libstdc++)
        set_target_properties(player_regression PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    endif()
elseif(WIN32)
    include(${CMAKE_CURRENT_SOURCE_DIR}/../windows/cmake/WindowsNative.cmake)
elseif(APPLE)
    enable_language(OBJCXX)
    set_target_properties(airplay_player PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        target_link_libraries(airplay_player PRIVATE "-framework AVFAudio")
    endif()
    target_sources(airplay_player PRIVATE backends/apple/apple_audio.mm backends/apple/apple_video.cpp backends/apple/apple_audio_decoder.cpp)
    target_link_libraries(airplay_player PRIVATE "-framework AudioToolbox" "-framework CoreAudio"
        "-framework CoreMedia" "-framework CoreVideo" "-framework VideoToolbox" "-framework CoreFoundation")
    set_target_properties(airplay_player PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(${CMAKE_CURRENT_SOURCE_DIR}/../linux/cmake/LinuxPlayer.cmake)
    airplay_configure_linux_player(airplay_player)
endif()
option(AIRPLAY_BUILD_TESTS "Build shared playback regressions" OFF)
if(AIRPLAY_BUILD_TESTS AND APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_testing()
    add_executable(player_tests tests/playback/main.cpp)
    target_include_directories(player_tests PRIVATE playback)
    target_link_libraries(player_tests PRIVATE airplay_player "-framework CoreVideo" "-framework CoreFoundation")
    add_test(NAME playback COMMAND player_tests)
    set_tests_properties(playback PROPERTIES LABELS "playback;backend")
    add_executable(session_tests tests/playback/session_test.cpp)
    target_include_directories(session_tests PRIVATE playback)
    target_link_libraries(session_tests PRIVATE airplay_player receiver_core "-framework CoreVideo" "-framework CoreFoundation")
    add_test(NAME sender_resume COMMAND session_tests)
    set_tests_properties(sender_resume PROPERTIES LABELS "protocol;playback;backend")
    add_test(NAME video_interruption COMMAND session_tests --video-interruption)
    set_tests_properties(video_interruption PROPERTIES LABELS "playback;backend")
endif()

include(cmake/ProductInputs.cmake)

if(AIRPLAY_BUILD_FFI AND (AIRPLAY_BUILD_TESTS OR AIRPLAY_WINDOWS_BUILD_TESTS))
    enable_testing()
    add_executable(receiver_control_tests tests/ffi/receiver_control_test.cpp)
    target_include_directories(receiver_control_tests PRIVATE playback "${AIRPLAY_DART_INCLUDE}")
    target_link_libraries(receiver_control_tests PRIVATE airplay_player receiver_core)
    add_test(NAME receiver_control COMMAND receiver_control_tests)
    set_tests_properties(receiver_control PROPERTIES LABELS "ffi")
endif()

if(AIRPLAY_BUILD_TESTS OR AIRPLAY_WINDOWS_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests/receiver)
endif()

if(AIRPLAY_BUILD_TESTS AND NOT WIN32 AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(protocol_rtp_tests tests/protocol/main.c)
    target_compile_definitions(protocol_rtp_tests PRIVATE PLIST_210 PLIST_230)
    if(CRYPTO_PREFIX)
        target_include_directories(protocol_rtp_tests PRIVATE "${CRYPTO_PREFIX}/include")
    endif()
    target_link_libraries(protocol_rtp_tests PRIVATE receiver_core)
    add_test(NAME rtp COMMAND protocol_rtp_tests)
    set_tests_properties(rtp PROPERTIES LABELS "protocol" TIMEOUT 30)
endif()
