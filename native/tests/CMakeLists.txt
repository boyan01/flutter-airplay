cmake_minimum_required(VERSION 3.22.1)
project(airplay_media_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
option(AIRPLAY_TEST_FFMPEG "Build the shared FFmpeg video adapter fixture" OFF)
option(AIRPLAY_SANITIZE "Enable address/undefined behavior checks" OFF)
if(AIRPLAY_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
if(AIRPLAY_TEST_FFMPEG)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec>=60 libavutil libswscale)
    add_executable(ffmpeg_video_tests
        ../backends/linux/linux_video.cpp ../backends/ffmpeg/ffmpeg_video.cpp
        ../../linux/tests/video_test.cpp)
    target_link_libraries(ffmpeg_video_tests PRIVATE PkgConfig::FFMPEG)
    add_test(NAME generate_video_fixtures COMMAND bash
        ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/generate_video_fixtures.sh ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
    set_tests_properties(generate_video_fixtures PROPERTIES FIXTURES_SETUP video-fixtures LABELS "fixture")
    add_test(NAME ffmpeg_video COMMAND ffmpeg_video_tests ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
    set_tests_properties(ffmpeg_video PROPERTIES FIXTURES_REQUIRED video-fixtures LABELS "playback;backend")
else()
    add_subdirectory(../../vendor/alac alac)
    add_executable(alac_tests playback/alac_test.cpp)
    target_link_libraries(alac_tests PRIVATE alac_decoder)
    add_test(NAME alac_decode COMMAND alac_tests)
    set_tests_properties(alac_decode PROPERTIES LABELS "playback;backend")
endif()
