cmake_minimum_required(VERSION 3.22.1)
project(android_receiver_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
option(HOST_SANITIZE "Enable address and undefined behavior checks" OFF)
if(HOST_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include(../app/src/main/cpp/ReceiverCore.cmake)
target_link_libraries(receiver_core PUBLIC pthread)
add_executable(receiver_test receiver_test.cpp receiver.cpp)
target_include_directories(receiver_test PRIVATE ../app/src/main/cpp)
target_link_libraries(receiver_test PRIVATE receiver_core)
add_executable(rtp_test ../../native/rtp-tests/main.c)
target_compile_definitions(rtp_test PRIVATE PLIST_210 PLIST_230)
if(CRYPTO_PREFIX)
    target_include_directories(rtp_test PRIVATE ${CRYPTO_PREFIX}/include)
endif()
target_link_libraries(rtp_test PRIVATE receiver_core)
